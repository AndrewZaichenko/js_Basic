/*
Вам необхідно вказати, яким буде результат операції та описати крок за кроком у вигляді коментарів, як відбувається перетворення в кожному прикладі:
*/

'number' + 3 + 3  // 'number33' => спочатку до строки 'number' додається чило 3 = 'number3', і потім ще раз -> 'number33'.

null + 3 // 3 => при додаванні null перетворюється на число 0.

5 && "qwerty" // "qwerty" => при && повертається або перший false операнд aбо останній якщо всі true, в нашому випадку 5 та "qwerty" є true.

+'40' + +'2' + "hillel"; // "42hillel" => спочатку унарний плюс перетворює '40' та '2' у numbers, потім складає їх, та додає string "hillel".

'10' - 5 === 6; // false => спочатку віднімаємо '10' - 5 = 5, потім порівнюємо тип та значення 5 === 6.

true + false // 1 => бінарний плюс перетворює true у 1, а false у 0.

'4px' - 3 // NaN => '4px' не перетворюється у number.

'4' - 3 // 1 => при відніманні string '4' перетворюється у number.

'6' + 3 ** 0; // '61' => спочатку 3 ** 0 = 1, а потім 1 додаємо до string '6'.

12 / '6' // 2 => при діленні string '6' перетворюється у number.

'10' + (5 === 6); // '10false' => спочатку виконується порівняння в дужках, а потім false додаємо до string '10'.

null == '' // false, тому що true тільки у випадку порівняння з undefined. 

3 ** (9 / 3); // 27 => спочатку виконується дія в дужках, потім 3 ** 3.

!!'false' == !!'true' // true => спочатку виконується перетворення у булеве значення за допомогою !!, а потім порівняння true == true.

0 || '0' && 1 // 1 => спочатку з '0' && 1 повертається 1 (тому що обидва true), потім з 0 || 1 повертається 1, як перше true значення.

(+null == false) < 1; // false => +null перетворюється у 0, (0 == false) це true, а true -> 1.

false && true || true // true => з false && true повертається false, як перше фолс значення, потім з false || true повертається true, як перше тру значення.

false && (false || true); // false => з (false || true) повертається true, як перше тру значення, потім з false && true повертається false, як перше фолс значення.

(+null == false) < 1 ** 5; // false => +null перетворюється у 0, (0 == false) це true, а true -> 1.
